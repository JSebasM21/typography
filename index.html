JavaScript
theme-control.js 
// NEXUS THEME CONTROL SYSTEM v2.1
class ThemeController {
    constructor() {
        this.init();
        this.initializeParticles();
    }
    
    toggleTheme() {
        const isDark = document.body.classList.toggle('dark-mode');
        this.updateInterface(isDark);
        this.savePreference(isDark);
        this.triggerTransitionEffects();
    }
    
    updateInterface(isDark) {
        document.getElementById('themeStatus').textContent = 
            isDark ? 'MODO OSCURO' : 'MODO CLARO';
        // Efectos visuales adicionales...
    }
}

NexusTheme utiliza tecnologías modernas para un control de tema óptimo:

  • LocalStorage API: Para persistencia de datos
